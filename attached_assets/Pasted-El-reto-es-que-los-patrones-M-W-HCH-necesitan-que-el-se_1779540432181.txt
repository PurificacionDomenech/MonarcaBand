pythondef calc_rsi_divergence(df, lookback=14):
    """
    Detecta divergencias RSI. Retorna coordenadas para dibujar en Plotly.
    """
    result = {
        "bear_div": False, "bull_div": False,
        "hidden_bear": False, "hidden_bull": False,
        "bear_lines": [],   # [(x0, y0_rsi, x1, y1_rsi, x0_price, y0_price, x1_price, y1_price)]
        "bull_lines": [],
    }
    if len(df) < lookback * 2 + 5:
        return result

    close = df["Close"]
    high  = df["High"]
    low   = df["Low"]
    rsi   = df["RSI"]
    if isinstance(close, pd.DataFrame): close = close.iloc[:, 0]
    if isinstance(high,  pd.DataFrame): high  = high.iloc[:, 0]
    if isinstance(low,   pd.DataFrame): low   = low.iloc[:, 0]
    if isinstance(rsi,   pd.DataFrame): rsi   = rsi.iloc[:, 0]

    n  = len(df)
    lb = max(3, min(lookback // 2, 5))

    def is_pivot_high(s, i, l):
        if i < l or i + l >= len(s): return False
        v = s.iloc[i]
        return all(s.iloc[i-j] <= v for j in range(1, l+1)) and \
               all(s.iloc[i+j] <= v for j in range(1, l+1))

    def is_pivot_low(s, i, l):
        if i < l or i + l >= len(s): return False
        v = s.iloc[i]
        return all(s.iloc[i-j] >= v for j in range(1, l+1)) and \
               all(s.iloc[i+j] >= v for j in range(1, l+1))

    search = min(lookback * 4, n - lb - 1)
    ph = [i for i in range(lb, lb + search) if is_pivot_high(rsi, i, lb)]
    pl = [i for i in range(lb, lb + search) if is_pivot_low(rsi,  i, lb)]

    # Divergencia bajista: precio HH, RSI LH
    if len(ph) >= 2:
        i1, i2 = ph[-2], ph[-1]
        if 3 <= (i2 - i1) <= lookback * 5:
            idx1, idx2 = n - 1 - i1, n - 1 - i2
            p_hh = float(high.iloc[idx1]) > float(high.iloc[idx2])
            r_lh = float(rsi.iloc[idx1])  < float(rsi.iloc[idx2])
            if p_hh and r_lh:
                result["bear_div"] = True
                result["bear_lines"].append({
                    "x0": df.index[idx2], "x1": df.index[idx1],
                    "rsi0": float(rsi.iloc[idx2]),  "rsi1": float(rsi.iloc[idx1]),
                    "p0":   float(high.iloc[idx2]), "p1":   float(high.iloc[idx1]),
                    "tipo": "regular_bear"
                })

    # Divergencia alcista: precio LL, RSI HL
    if len(pl) >= 2:
        i1, i2 = pl[-2], pl[-1]
        if 3 <= (i2 - i1) <= lookback * 5:
            idx1, idx2 = n - 1 - i1, n - 1 - i2
            p_ll = float(low.iloc[idx1]) < float(low.iloc[idx2])
            r_hl = float(rsi.iloc[idx1]) > float(rsi.iloc[idx2])
            if p_ll and r_hl:
                result["bull_div"] = True
                result["bull_lines"].append({
                    "x0": df.index[idx2], "x1": df.index[idx1],
                    "rsi0": float(rsi.iloc[idx2]), "rsi1": float(rsi.iloc[idx1]),
                    "p0":   float(low.iloc[idx2]), "p1":   float(low.iloc[idx1]),
                    "tipo": "regular_bull"
                })

    # Divergencia bajista oculta: precio LH, RSI HH
    if len(ph) >= 2:
        i1, i2 = ph[-2], ph[-1]
        if 3 <= (i2 - i1) <= lookback * 5:
            idx1, idx2 = n - 1 - i1, n - 1 - i2
            p_lh = float(high.iloc[idx1]) < float(high.iloc[idx2])
            r_hh = float(rsi.iloc[idx1])  > float(rsi.iloc[idx2])
            if p_lh and r_hh:
                result["hidden_bear"] = True
                result["bear_lines"].append({
                    "x0": df.index[idx2], "x1": df.index[idx1],
                    "rsi0": float(rsi.iloc[idx2]),  "rsi1": float(rsi.iloc[idx1]),
                    "p0":   float(high.iloc[idx2]), "p1":   float(high.iloc[idx1]),
                    "tipo": "hidden_bear"
                })

    # Divergencia alcista oculta: precio HL, RSI LL
    if len(pl) >= 2:
        i1, i2 = pl[-2], pl[-1]
        if 3 <= (i2 - i1) <= lookback * 5:
            idx1, idx2 = n - 1 - i1, n - 1 - i2
            p_hl = float(low.iloc[idx1]) > float(low.iloc[idx2])

def calc_pattern_mw(df, lookback=40, tol=0.018):
    """
    Detecta M (doble techo) y W (doble suelo).
    Retorna coordenadas completas + estado (confirmado/formando).
    'formando' = P2/V2 está dentro de la zona del primero pero aún no ha roto neckline.
    """
    result = {"M": None, "W": None}
    if len(df) < 15:
        return result

    high  = df["High"]
    low   = df["Low"]
    close = df["Close"]
    rsi   = df["RSI"]
    if isinstance(high,  pd.DataFrame): high  = high.iloc[:, 0]
    if isinstance(low,   pd.DataFrame): low   = low.iloc[:, 0]
    if isinstance(close, pd.DataFrame): close = close.iloc[:, 0]
    if isinstance(rsi,   pd.DataFrame): rsi   = rsi.iloc[:, 0]

    window_df = df.iloc[-lookback:]
    wh = high.iloc[-lookback:]
    wl = low.iloc[-lookback:]
    wr = rsi.iloc[-lookback:]
    wc = close.iloc[-lookback:]
    dates = list(window_df.index)
    n = len(window_df)

    def local_highs(series, min_dist=3):
        peaks = []
        for i in range(2, n - 2):
            v = float(series.iloc[i])
            if (v >= float(series.iloc[i-1]) and v >= float(series.iloc[i-2]) and
                v >= float(series.iloc[i+1]) and v >= float(series.iloc[i+2])):
                if not peaks or (i - peaks[-1][0]) >= min_dist:
                    peaks.append((i, v))
        return peaks

    def local_lows(series, min_dist=3):
        troughs = []
        for i in range(2, n - 2):
            v = float(series.iloc[i])
            if (v <= float(series.iloc[i-1]) and v <= float(series.iloc[i-2]) and
                v <= float(series.iloc[i+1]) and v <= float(series.iloc[i+2])):
                if not troughs or (i - troughs[-1][0]) >= min_dist:
                    troughs.append((i, v))
        return troughs

    highs  = local_highs(wh)
    lows   = local_lows(wl)

    # ── Patrón M ─────────────────────────────────────────
    if len(highs) >= 2:
        for i in range(len(highs) - 1):
            h1_i, h1_v = highs[i]
            h2_i, h2_v = highs[-1]
            if h2_i - h1_i < 5:
                continue
            if abs(h1_v - h2_v) / h1_v <= tol:
                # Neckline: mínimo entre los dos picos
                mid_lows = [lows[j] for j in range(len(lows))
                            if h1_i < lows[j][0] < h2_i]
                neckline_v = min((v for _, v in mid_lows), default=None)
                if neckline_v is None:
                    neckline_v = float(wl.iloc[h1_i:h2_i].min())

                rsi1 = float(wr.iloc[h1_i])
                rsi2 = float(wr.iloc[h2_i])
                current_price = float(wc.iloc[-1])

                # Estado
                if current_price < neckline_v:
                    estado = "confirmado"
                elif abs(current_price - h2_v) / h2_v <= tol * 1.5:
                    estado = "formando_p2"
                else:
                    estado = "formando"

                result["M"] = {
                    "p1_x": dates[h1_i], "p1_y": h1_v,
                    "p2_x": dates[h2_i], "p2_y": h2_v,
                    "neckline": neckline_v,
                    "rsi1": rsi1, "rsi2": rsi2,
                    "rsi_div": rsi2 < rsi1 - 2,
                    "estado": estado,
                    "bearish": True,
                }
                break

    # ── Patrón W ─────────────────────────────────────────
    if len(lows) >= 2:
        for i in range(len(lows) - 1):
            l1_i, l1_v = lows[i]
            l2_i, l2_v = lows[-1]
            if l2_i - l1_i < 5:
                continue
            if abs(l1_v - l2_v) / l1_v <= tol:
                mid_highs = [highs[j] for j in range(len(highs))
                             if l1_i < highs[j][0] < l2_i]
                neckline_v = max((v for _, v in mid_highs), default=None)
                if neckline_v is None:
                    neckline_v = float(wh.iloc[l1_i:l2_i].max())

                rsi1 = float(wr.iloc[l1_i])
                rsi2 = float(wr.iloc[l2_i])
                current_price = float(wc.iloc[-1])

                if current_price > neckline_v:
                    estado = "confirmado"
                elif abs(current_price - l2_v) / l2_v <= tol * 1.5:
                    estado = "formando_v2"
                else:
                    estado = "formando"

                result["W"] = {
                    "v1_x": dates[l1_i], "v1_y": l1_v,
                    "v2_x": dates[l2_i], "v2_y": l2_v,
                    "neckline": neckline_v,
                    "rsi1": rsi1, "rsi2": rsi2,
                    "rsi_div": rsi2 > rsi1 + 2,
                    "estado": estado,
                    "bullish": True,
                }
                break

    return result


def calc_pattern_hch(df, lookback=60, tol=0.02):
    """
    Detecta HCH (bajista) e HCH invertido (alcista).
    Estado 'formando_hd' = hombro derecho aún no completado.
    """
    result = {"HCH": None, "HCH_inv": None}
    if len(df) < 20:
        return result

    high  = df["High"]
    low   = df["Low"]
    close = df["Close"]
    rsi   = df["RSI"]
    if isinstance(high,  pd.DataFrame): high  = high.iloc[:, 0]
    if isinstance(low,   pd.DataFrame): low   = low.iloc[:, 0]
    if isinstance(close, pd.DataFrame): close = close.iloc[:, 0]
    if isinstance(rsi,   pd.DataFrame): rsi   = rsi.iloc[:, 0]

    wh = high.iloc[-lookback:]
    wl = low.iloc[-lookback:]
    wr = rsi.iloc[-lookback:]
    wc = close.iloc[-lookback:]
    dates = list(df.index[-lookback:])
    n = len(wh)

    def find_peaks(series, min_dist=4):
        peaks = []
        for i in range(3, n - 3):
            v = float(series.iloc[i])
            if all(v >= float(series.iloc[i-j]) for j in range(1, 4)) and \
               all(v >= float(series.iloc[i+j]) for j in range(1, 4)):
                if not peaks or (i - peaks[-1][0]) >= min_dist:
                    peaks.append((i, v, dates[i]))
        return peaks

    def find_troughs(series, min_dist=4):
        troughs = []
        for i in range(3, n - 3):
            v = float(series.iloc[i])
            if all(v <= float(series.iloc[i-j]) for j in range(1, 4)) and \
               all(v <= float(series.iloc[i+j]) for j in range(1, 4)):
                if not troughs or (i - troughs[-1][0]) >= min_dist:
                    troughs.append((i, v, dates[i]))
        return troughs

    peaks   = find_peaks(wh)
    troughs = find_troughs(wl)
    current = float(wc.iloc[-1])

    # ── HCH bajista: HI < CAB > HD, con HI ≈ HD ─────────
    if len(peaks) >= 3:
        for i in range(len(peaks) - 2):
            hi_i, hi_v, hi_x = peaks[i]
            cab_i, cab_v, cab_x = peaks[i+1]
            hd_i, hd_v, hd_x = peaks[-1]

            if cab_v <= hi_v or cab_v <= hd_v:
                continue
            if abs(hi_v - hd_v) / hi_v > tol * 2:
                continue
            if hd_i - cab_i < 5:
                continue

            # Neckline: recta entre los dos valles entre HI-CAB y CAB-HD
            v1 = [t for t in troughs if hi_i < t[0] < cab_i]
            v2 = [t for t in troughs if cab_i < t[0] < hd_i]
            if not v1 or not v2:
                continue
            nk1_i, nk1_v, nk1_x = min(v1, key=lambda x: x[1])
            nk2_i, nk2_v, nk2_x = min(v2, key=lambda x: x[1])

            if current < nk2_v:
                estado = "confirmado"
            elif abs(current - hd_v) / hd_v <= tol:
                estado = "formando_hd"
            else:
                estado = "formando"

            result["HCH"] = {
                "hi_x": hi_x,   "hi_y": hi_v,
                "cab_x": cab_x, "cab_y": cab_v,
                "hd_x": hd_x,   "hd_y": hd_v,
                "nk1_x": nk1_x, "nk1_y": nk1_v,
                "nk2_x": nk2_x, "nk2_y": nk2_v,
                "neckline_slope": (nk2_v - nk1_v) / max(nk2_i - nk1_i, 1),
                "rsi_hi": float(wr.iloc[hi_i]),
                "rsi_cab": float(wr.iloc[cab_i]),
                "rsi_hd": float(wr.iloc[hd_i]),
                "estado": estado,
                "bearish": True,
            }
            break

    # ── HCH invertido alcista: hi < cab < hd (invertido) ─
    if len(troughs) >= 3:
        for i in range(len(troughs) - 2):
            hi_i, hi_v, hi_x = troughs[i]
            cab_i, cab_v, cab_x = troughs[i+1]
            hd_i, hd_v, hd_x = troughs[-1]

            if cab_v >= hi_v or cab_v >= hd_v:
                continue
            if abs(hi_v - hd_v) / hi_v > tol * 2:
                continue
            if hd_i - cab_i < 5:
                continue

            p1 = [t for t in peaks if hi_i < t[0] < cab_i]
            p2 = [t for t in peaks if cab_i < t[0] < hd_i]
            if not p1 or not p2:
                continue
            nk1_i, nk1_v, nk1_x = max(p1, key=lambda x: x[1])
            nk2_i, nk2_v, nk2_x = max(p2, key=lambda x: x[1])

            if current > nk2_v:
                estado = "confirmado"
            elif abs(current - hd_v) / hd_v <= tol:
                estado = "formando_hd"
            else:
                estado = "formando"

            result["HCH_inv"] = {
                "hi_x": hi_x,   "hi_y": hi_v,
                "cab_x": cab_x, "cab_y": cab_v,
                "hd_x": hd_x,   "hd_y": hd_v,
                "nk1_x": nk1_x, "nk1_y": nk1_v,
                "nk2_x": nk2_x, "nk2_y": nk2_v,
                "rsi_hi": float(wr.iloc[hi_i]),
                "rsi_cab": float(wr.iloc[cab_i]),
                "rsi_hd": float(wr.iloc[hd_i]),
                "estado": estado,
                "bullish": True,
            }
            break

    return result


def calc_candle_patterns(df):
    if len(df) < 3:
        return {}
    o  = float(df["Open"].iloc[-1])
    h  = float(df["High"].iloc[-1])
    l  = float(df["Low"].iloc[-1])
    c  = float(df["Close"].iloc[-1])
    o2 = float(df["Open"].iloc[-2])
    c2 = float(df["Close"].iloc[-2])
    body    = abs(c - o)
    rng     = h - l if h > l else 0.0001
    upper_w = h - max(c, o)
    lower_w = min(c, o) - l
    body_p  = body / rng
    patterns = {}
    if c2 < o2 and c > o and c > o2 and o < c2:
        patterns["engulfing_bull"] = {"tipo":"bullish","desc":"Engulfing alcista"}
    if c2 > o2 and c < o and c < o2 and o > c2:
        patterns["engulfing_bear"] = {"tipo":"bearish","desc":"Engulfing bajista"}
    if upper_w >= body * 2 and lower_w <= body * 0.5 and body_p < 0.35:
        patterns["shooting_star"] = {"tipo":"bearish","desc":"Shooting star"}
    if lower_w >= body * 2 and upper_w <= body * 0.5 and body_p < 0.35:
        patterns["hammer"] = {"tipo":"bullish","desc":"Hammer"}
    if lower_w >= body * 3 and upper_w <= body:
        patterns["pinbar_bull"] = {"tipo":"bullish","desc":"Pin bar alcista"}
    if upper_w >= body * 3 and lower_w <= body:
        patterns["pinbar_bear"] = {"tipo":"bearish","desc":"Pin bar bajista"}
    if c > o and upper_w <= body * 0.05 and lower_w <= body * 0.05 and body_p > 0.85:
        patterns["marubozu_bull"] = {"tipo":"bullish","desc":"Marubozu alcista"}
    if c < o and upper_w <= body * 0.05 and lower_w <= body * 0.05 and body_p > 0.85:
        patterns["marubozu_bear"] = {"tipo":"bearish","desc":"Marubozu bajista"}
    return patterns
