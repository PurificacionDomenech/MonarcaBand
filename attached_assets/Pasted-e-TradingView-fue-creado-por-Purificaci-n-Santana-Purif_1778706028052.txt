// ==================== FUNCIÓN MA ====================
getMA(series float src, simple int length, simple string maType) =>
    switch maType
        "EMA"    => ta.ema(src, length)
        "HMA"    => ta.hma(src, length)
        "SMA"    => ta.sma(src, length)
        "TMA"    => ta.sma(ta.sma(src, length), length)
        "WMA"    => ta.wma(src, length)
        "DEMA"   => 2 * ta.ema(src, length) - ta.ema(ta.ema(src, length), length)
        "TEMA"   => 3 * ta.ema(src, length) - 3 * ta.ema(ta.ema(src, length), length) + ta.ema(ta.ema(ta.ema(src, length), length), length)
        "VWMA"   => ta.vwma(src, length)
        "ZLEMA"  => ta.ema(src + (src - src[1 + math.floor((length - 1) / 2)]), length)
        "LinReg" => ta.linreg(src, length, 0)
        =>          ta.sma(src, length)

// ==================== FUNCIÓN AUXILIAR RANGO ====================
_inRange(bool cond, int rMin, int rMax) =>
    bars = ta.barssince(cond)
    rMin <= bars and bars <= rMax

// ==================== CÁLCULO DIVERGENCIAS (sin line.new dentro) ====================
f_divs(int lbL, int lbR, int rMin, int rMax) =>
    float rsiLBR  = rsi[lbR]
    float lowLBR  = low[lbR]
    float highLBR = high[lbR]

    bool plFound = not na(ta.pivotlow(rsi,  lbL, lbR))
    bool phFound = not na(ta.pivothigh(rsi, lbL, lbR))

    // Regular Alcista: precio LL, RSI HL
    bool bullC = plFound and
                 lowLBR  < ta.valuewhen(plFound, lowLBR,  1) and
                 rsiLBR  > ta.valuewhen(plFound, rsiLBR,  1) and
                 _inRange(plFound[1], rMin, rMax)

    // Regular Bajista: precio HH, RSI LH
    bool bearC = phFound and
                 highLBR > ta.valuewhen(phFound, highLBR, 1) and
                 rsiLBR  < ta.valuewhen(phFound, rsiLBR,  1) and
                 _inRange(phFound[1], rMin, rMax)

    // Oculta Alcista: precio HL, RSI LL
    bool hBullC = plFound and
                  lowLBR  > ta.valuewhen(plFound, lowLBR,  1) and
                  rsiLBR  < ta.valuewhen(plFound, rsiLBR,  1) and
                  _inRange(plFound[1], rMin, rMax)

    // Oculta Bajista: precio LH, RSI HH
    bool hBearC = phFound and
                  highLBR < ta.valuewhen(phFound, highLBR, 1) and
                  rsiLBR  > ta.valuewhen(phFound, rsiLBR,  1) and
                  _inRange(phFound[1], rMin, rMax)

    [rsiLBR, lowLBR, highLBR, plFound, phFound, bullC, bearC, hBullC, hBearC]

// ==================== CALCULAR LOS 3 NIVELES ====================
